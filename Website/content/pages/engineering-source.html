Eigen::Vector3d LegController::forward_kinematic(
    const Eigen::Vector3d &joint_angles) const
{ // clang-format off
    double qh = joint_angles(0);
    double qt = joint_angles(1);
    double qs = joint_angles(2);
    double ls = geometry.shank_length;
    double lt = geometry.thigh_length;
    double lh = geometry.hip_length;

    double x = -ls * std::sin(qs - qt) 
        + lt * std::sin(qt);
    double y = ls * std::sin(qs) * std::sin(qh) * std::sin(qt) 
        + ls * std::cos(qs) * std::sin(qh) * std::cos(qt) 
        + lt * std::cos(qt) * std::sin(qh) 
        + lh * cos(qh);
    double z = ls * std::sin(qs) * std::cos(qh) * std::sin(qt) 
        + ls * std::cos(qs) * std::cos(qh) * std::cos(qt) 
        + lt * std::cos(qt) * std::cos(qh) 
        - lh * std::sin(qh);
    Eigen::Vector3d foot_position_leg_frame;
    foot_position_leg_frame << x,
        y,
        z;
    return foot_position_leg_frame;
}

Eigen::Vector3d LegController::inverse_kinematic(
    const Eigen::Vector3d &foot_positions) const
{
    double x = foot_positions(0);
    double y = foot_positions(1);
    double z = foot_positions(2);
    double ls = geometry.shank_length;
    double lt = geometry.thigh_length;
    double lh = geometry.hip_length;
    if (!foot_positions.allFinite() || y * y + z * z < lh * lh)
    {
        std::cerr << "unreachable target, invalid hip position\n";
        return invalidVector();
    }
    double a = std::sqrt(y * y + z * z - lh * lh);
    if ((x == 0 && a == 0) || (lh == 0 && a == 0))
    {
        std::cerr << "indeterminate target, invalid joint angles\n";
        return invalidVector();
    }
    double qh = std::atan2(a * y - lh * z, lh * y + a * z);
    double c = (x * x + a * a - lt * lt - ls * ls) / (2 * ls * lt);
    if (!std::isfinite(c) || std::abs(c) > 1 + 1e-12)
    {
        std::cerr << "unreachable target, invalid shank angle\n";
        return invalidVector();
    }
    double qs = std::acos(std::clamp(c, -1.0, 1.0));
    double b = lt + ls * std::cos(qs);
    double d = ls * std::sin(qs);
    double qt = std::atan2(b * x + d * a, b * a - d * x);
    Eigen::Vector3d joint_angles;
    joint_angles << qh,
        qt,
        qs;

    return joint_angles;
}

Eigen::Matrix3d LegController::forward_jacobian_transform(
    const Eigen::Vector3d &joint_angles) const
{ // clang-format off
    double qh = joint_angles(0);
    double qt = joint_angles(1);
    double qs = joint_angles(2);
    double ls = geometry.shank_length;
    double lt = geometry.thigh_length;
    double lh = geometry.hip_length;
    double x = -ls * std::sin(qs - qt)
        + lt * std::sin(qt);
    double y = ls * std::sin(qs) * std::sin(qh) * std::sin(qt)
        + ls * std::cos(qs) * std::sin(qh) * std::cos(qt)
        + lt * std::cos(qt) * std::sin(qh)
        + lh * cos(qh);
    double z = ls * std::sin(qs) * std::cos(qh) * std::sin(qt)
        + ls * std::cos(qs) * std::cos(qh) * std::cos(qt)
        + lt * std::cos(qt) * std::cos(qh)
        - lh * std::sin(qh);
    Eigen::Matrix3d forward_jacobian;
    forward_jacobian <<  0, lt * std::cos(qt) + ls * std::cos(qs - qt), -ls * std::cos(qs - qt),
                        z, -x * std::sin(qh), -ls * std::sin(qs - qt) * std::sin(qh),
                       -y, -x * std::cos(qh), -ls * std::sin(qs - qt) * std::cos(qh);
    return forward_jacobian;
}
//clang-format on

Eigen::Matrix3d LegController::get_damped_inverse_J(
    const Eigen::Matrix3d &forward_jacobian) const
{
    const Eigen::Matrix3d jacobian_transpose =
        forward_jacobian.transpose();
    const Eigen::Matrix3d normal_matrix =
        jacobian_transpose * forward_jacobian +
        jacobian_damping * jacobian_damping *
            Eigen::Matrix3d::Identity();
    return normal_matrix.ldlt().solve(jacobian_transpose);

bool LegController::calculateJointVelocity()
{
    const auto &foot_velocity = leg_command.foot_velocity_command_vector;
    auto &joint_velocity = leg_command.joint_velocity_command_vector;
    if (!foot_velocity.allFinite() || foot_velocity.norm() > max_foot_velocity)
    {
        std::cerr << getName() << ": invalid or excessive foot velocity [" << foot_velocity.transpose()
                  << "] m/s; maximum norm=" << max_foot_velocity << " m/s\n";
        return false;
    }
    const Eigen::Matrix3d jacobian = forward_jacobian_transform(leg_state.joint_position_state_vector);
    if (!validateJacobian(jacobian))
        return false;
    joint_velocity = get_damped_inverse_J(jacobian) * foot_velocity;
    return joint_velocity.allFinite();
}

bool LegController::setFootTargetPDO3()
{
    leg_command.joint_feedforward_torque_command_vector.setZero(); // Cartesian trajectory path has no explicit FF input.
    leg_command.joint_position_command_vector = inverse_kinematic(leg_command.foot_position_command_vector);
    return calculateJointVelocity() && stageJointCommandsPDO3();
}

bool LegController::setJointTargetPDO3()
{
    leg_command.foot_position_command_vector = forward_kinematic(leg_command.joint_position_command_vector);
    leg_command.foot_velocity_command_vector =
        forward_jacobian_transform(leg_state.joint_position_state_vector) * leg_command.joint_velocity_command_vector;
    return stageJointCommandsPDO3();
}

bool LegController::stageJointCommandsPDO3()
{
    if (!validateLegCommands())
        return false;
    // Validate all joints before staging any; no CAN transmission here.
    for (std::size_t i = 0; i < joints_per_leg; ++i)
    {
        auto &joint = joint_controllers[i];
        joint.joint_command = {static_cast<float>(leg_command.joint_position_command_vector[i]),
                               static_cast<float>(leg_command.joint_velocity_command_vector[i]),
                               static_cast<float>(leg_command.joint_feedforward_torque_command_vector[i])};
        if (!joint.setTargetPDO3())
            return false;
    }
    return true;

bool JointController::validateJointCommands(const JointCommand &command) const noexcept
{
    if (!hasInitialReference() || !validateJointStates())
        return false;

    if (!std::isfinite(command.position) ||
        command.position < safety_limits.position_lower ||
        command.position > safety_limits.position_upper)
    {
        std::cerr << getName() << ": absolute position target outside limits\n";
        return false;
    }
    if (!withinAbsoluteLimit(command.position - joint_state.position, safety_limits.position_error))
    {
        std::cerr << getName() << ": position command tracking error exceeded\n";
        return false;
    }
    if (!withinAbsoluteLimit(command.velocity, safety_limits.velocity))
    {
        std::cerr << getName() << ": velocity target outside limits\n";
        return false;
    }
    if (!withinAbsoluteLimit(command.feedforward_torque, safety_limits.feedforward_torque_limit))
    {
        std::cerr << getName() << ": feedforward torque outside limits\n";
        return false;
    }
    return true;
}
